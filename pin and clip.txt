    <!-- Scripts -->
    <script src="https://cdn.jsdelivr.net/npm/@tensorflow/tfjs@latest/dist/tf.min.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/@teachablemachine/image@latest/dist/teachablemachine-image.min.js"></script>
    
    <script type="text/javascript">
        const URL = "https://teachablemachine.withgoogle.com/models/hxs6Yifqy/";

        let model, webcam, labelContainer, maxPredictions;
        let predictInterval; // ตัวแปรสำหรับเก็บ Interval 3 วินาที

        async function init() {
            // ปิดปุ่มกดหลังจากเริ่มทำงานเพื่อป้องกันการกดซ้ำ
            const startBtn = document.getElementById("start-btn");
            startBtn.disabled = true;
            startBtn.innerText = "กำลังโหลดโมเดล...";

            const modelURL = URL + "model.json";
            const metadataURL = URL + "metadata.json";

            // โหลด Model
            model = await tmImage.load(modelURL, metadataURL);
            maxPredictions = model.getTotalClasses();

            // ตั้งค่า Webcam (ปรับขนาดให้แสดงผลพอดีกับกรอบ)
            const flip = true; 
            webcam = new tmImage.Webcam(250, 250, flip); 
            await webcam.setup(); 
            await webcam.play();
            
            // รัน loop เพื่อให้ภาพในกล้องเคลื่อนไหวลื่นไหลเป็นปกติ
            window.requestAnimationFrame(loop);

            // เคลียร์ข้อความเริ่มต้นในกรอบเว็บบอร์ด แล้วนำกล้องไปใส่
            const webcamContainer = document.getElementById("webcam-container");
            webcamContainer.innerHTML = "";
            webcamContainer.appendChild(webcam.canvas);

            // สร้างโครงสร้าง Progress Bar สำหรับแสดงผลแยกตามคลาส
            labelContainer = document.getElementById("label-container");
            labelContainer.innerHTML = ""; // ล้างข้อมูลเก่า
            for (let i = 0; i < maxPredictions; i++) {
                // สร้างกล่องแถว
                const row = document.createElement("div");
                row.className = "prediction-row";

                // สร้างข้อความประกอบ (ชื่อคลาส และ %)
                const info = document.createElement("div");
                info.className = "prediction-info";
                info.innerHTML = `<span class="class-name">-</span><span class="class-prob">0%</span>`;
                
                // สร้างแถบ Progress Bar
                const barBg = document.createElement("div");
                barBg.className = "progress-bar-bg";
                const barFill = document.createElement("div");
                barFill.className = "progress-bar-fill";
                
                barBg.appendChild(barFill);
                row.appendChild(info);
                row.appendChild(barBg);
                labelContainer.appendChild(row);
            }

            startBtn.innerText = "ระบบกำลังทำงาน...";
            document.getElementById("status-indicator").innerText = "เริ่มการวิเคราะห์ทุกๆ 3 วินาที";

            // เรียกให้วิเคราะห์ครั้งแรกทันที
            predict();

            // ⚡ ส่วนสำคัญ: สั่งให้ฟังก์ชัน predict ทำงานทุกๆ 3000 มิลลิวินาที (3 วินาที)
            predictInterval = setInterval(async () => {
                document.getElementById("status-indicator").innerText = "กำลังวิเคราะห์รูปภาพ... 🔍";
                await predict();
                setTimeout(() => {
                    document.getElementById("status-indicator").innerText = "รอรอบถัดไป (ทุกๆ 3 วินาที)";
                }, 500); // แสดงสถานะกำลังวิเคราะห์แป๊บหนึ่งแล้วเปลี่ยนกลับ
            }, 3000);
        }

        async function loop() {
            webcam.update(); // อัปเดตเฟรมของกล้องให้เคลื่อนไหวตลอดเวลา (Real-time ลื่นไหล)
            window.requestAnimationFrame(loop);
        }

        // ฟังก์ชันวิเคราะห์ที่จะถูกเรียกทุกๆ 3 วินาที
        async function predict() {
            if (!model || !webcam) return;

            const prediction = await model.predict(webcam.canvas);
            
            for (let i = 0; i < maxPredictions; i++) {
                const className = prediction[i].className;
                const probability = prediction[i].probability;
                const percentage = (probability * 100).toFixed(0) + "%";

                // อัปเดตข้อความและเปอร์เซ็นต์
                const row = labelContainer.childNodes[i];
                row.querySelector(".class-name").innerText = className;
                row.querySelector(".class-prob").innerText = percentage;

                // อัปเดตความยาวของแถบสี (Progress Bar)
                row.querySelector(".progress-bar-fill").style.width = percentage;
            }
        }
