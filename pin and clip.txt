<script type="text/javascript">
    const URL = "https://teachablemachine.withgoogle.com/models/hxs6Yifqy/";

    let model, webcam, labelContainer, maxPredictions;
    let millisecondsLeft = 3000; 
    const intervalPeriod = 3000; 
    const clockSpeed = 100; // อัปเดต UI ทุกๆ 0.1 วินาทีเพื่อความสมูท
    let clockInterval = null;

    async function init() {
        const startBtn = document.getElementById("btn-init");
        startBtn.disabled = true;
        startBtn.innerText = "LOADING MODEL...";

        try {
            model = await tmImage.load(URL + "model.json", URL + "metadata.json");
            maxPredictions = model.getTotalClasses();

            webcam = new tmImage.Webcam(200, 200, true); 
            await webcam.setup();
            await webcam.play();

            document.getElementById("webcam-container").innerHTML = "";
            labelContainer = document.getElementById("label-container");
            labelContainer.innerHTML = "";

            document.getElementById("webcam-container").appendChild(webcam.canvas);

            for (let i = 0; i < maxPredictions; i++) {
                const card = document.createElement("div");
                card.className = "class-card";
                card.id = `class-box-${i}`;
                card.innerHTML = `
                    <div class="card-info">
                        <span class="card-name" id="txt-name-${i}">Connecting...</span>
                        <span class="card-percent" id="txt-percent-${i}">0%</span>
                    </div>
                    <div class="track-bar">
                        <div class="fill-bar" id="bar-fill-${i}"></div>
                    </div>
                `;
                labelContainer.appendChild(card);
            }

            startBtn.innerText = "SYSTEM ACTIVE";
            window.requestAnimationFrame(loop);

            // รันการทายใจครั้งแรกทันทีตอนเริ่มระบบ
            await predict();
            startEngineClock();

        } catch (error) {
            console.error("Initialization Failed: ", error);
            startBtn.disabled = false;
            startBtn.innerText = "RETRY SETUP";
            alert("ไม่สามารถเปิดใช้งานโมเดลหรือกล้องได้ กรุณาตรวจสอบสิทธิ์การใช้กล้อง");
        }
    }

    async function loop() {
        webcam.update();
        window.requestAnimationFrame(loop);
    }

    function startEngineClock() {
        if(clockInterval) clearInterval(clockInterval);
        
        const timeRing = document.getElementById("time-ring");
        const strokeMaxOffset = 660; // 2 * PI * r (2 * 3.14159 * 105)

        clockInterval = setInterval(async () => {
            millisecondsLeft -= clockSpeed;

            if (millisecondsLeft <= 0) {
                millisecondsLeft = intervalPeriod; 
                await predict(); // เรียก AI วิเคราะห์ทุกๆ 3 วินาทีเมื่อตัวนับหมดเวลา
            }

            // คำนวณความยาววงกลมให้สัมพันธ์กับเวลาที่เหลือ
            const updatedOffset = strokeMaxOffset - (strokeMaxOffset * (millisecondsLeft / intervalPeriod));
            timeRing.style.strokeDashoffset = updatedOffset;

            document.getElementById("timer-countdown").innerText = `NEXT CAPTURE: ${(millisecondsLeft / 1000).toFixed(1)}s`;
        }, clockSpeed);
    }

    async function predict() {
        if (!model || !webcam) return;

        const prediction = await model.predict(webcam.canvas);
        
        let maximumProbability = -1;
        let indexOfTheWinner = -1;

        // หาผลลัพธ์ที่มีเปอร์เซ็นต์สูงสุด
        for (let i = 0; i < maxPredictions; i++) {
            if (prediction[i].probability > maximumProbability) {
                maximumProbability = prediction[i].probability;
                indexOfTheWinner = i;
            }
        }

        // แสดงผลลัพธ์และอัปเดต UI แถบพลัง
        for (let i = 0; i < maxPredictions; i++) {
            const nameOfClass = prediction[i].className;
            const probabilityValue = prediction[i].probability;
            const roundedPercentage = Math.round(probabilityValue * 100);

            document.getElementById(`txt-name-${i}`).innerText = nameOfClass;
            document.getElementById(`txt-percent-${i}`).innerText = roundedPercentage + "%";
            document.getElementById(`bar-fill-${i}`).style.width = roundedPercentage + "%";

            const cardBlock = document.getElementById(`class-box-${i}`);
            
            // ไฮไลท์การ์ดสีเขียวแบบเรืองแสงเมื่อเป็น Class ที่ชนะ และมีมากกว่า 40%
            if (i === indexOfTheWinner && roundedPercentage >= 40) {
                cardBlock.classList.add("active-match");
            } else {
                cardBlock.classList.remove("active-match");
            }
        }
    }
</script>
