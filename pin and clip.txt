<script type="text/javascript">
    // ลิงก์โมเดลปัจจุบันตัวล่าสุดของคุณ
    const URL = "https://teachablemachine.withgoogle.com/models/hxs6Yifqy/";[cite: 2]

    let model, webcam, labelContainer, maxPredictions;
    let millisecondsLeft = 3000; 
    const intervalPeriod = 3000; // 3 วินาที
    const clockSpeed = 100; // รีเฟรชหลอด UI ทุก 0.1 วินาที

    async function init() {
        const startBtn = document.getElementById("btn-init");
        startBtn.disabled = true;
        startBtn.innerText = "LOADING MODEL...";

        try {
            // โหลดโมเดล
            model = await tmImage.load(URL + "model.json", URL + "metadata.json");[cite: 2]
            maxPredictions = model.getTotalClasses();[cite: 2]

            // เปิดใช้งานเว็บแคม (ปรับเป็น 230x230 เพื่อให้รับกับการจัดวางภาพแบบทรงกลม)
            webcam = new tmImage.Webcam(230, 230, true);[cite: 2]
            await webcam.setup();[cite: 2]
            await webcam.play();[cite: 2]

            // เคลียร์พื้นที่หน้าจอเดิม
            document.getElementById("webcam-container").innerHTML = "";
            labelContainer = document.getElementById("label-container");
            labelContainer.innerHTML = "";

            // แนบ Canvas ของวิดีโอเข้าไป
            document.getElementById("webcam-container").appendChild(webcam.canvas);[cite: 2]

            // วาดโครงสร้างการแสดงแถบเปอร์เซ็นต์ (Class Cards)
            for (let i = 0; i < maxPredictions; i++) {[cite: 2]
                const card = document.createElement("div");
                card.className = "class-card";
                card.id = `class-box-${i}`;
                card.innerHTML = `
                    <div class="card-info">
                        <span class="card-name" id="txt-name-${i}">Connecting...</span>
                        <span class="card-percent" id="txt-percent-${i}">0%</span>
                    </div>
                    <div class="track-bar">
                        <div class="fill-bar" id="bar-fill-${i}"></div>
                    </div>
                `;
                labelContainer.appendChild(card);[cite: 2]
            }

            startBtn.innerText = "SYSTEM ACTIVE";
            
            // สั่งรันลูปสำหรับการแสดงผลวิดีโอของเว็บแคมให้ลื่นไหลแบบ Realtime
            window.requestAnimationFrame(loop);[cite: 2]

            // ทำการทำนายภาพครั้งแรกทันทีตอนเริ่มระบบ
            await predict();[cite: 2]

            // สตาร์ทตัวนับเวลาถอยหลัง 3 วินาทีแบบแม่นยำสูง
            startEngineClock();

        } catch (error) {
            console.error("Initialization Failed: ", error);
            startBtn.disabled = false;
            startBtn.innerText = "RETRY SETUP";
            alert("ไม่สามารถเปิดใช้งานโมเดลหรือกล้องได้ โปรดอนุญาตให้บราวเซอร์ใช้งานสิทธิ์กล้องถ่ายภาพ");
        }
    }

    async function loop() {
        webcam.update(); // อัปเดตเฟรมภาพเคลื่อนไหวบนเว็บแคมให้ลื่นไหล[cite: 2]
        window.requestAnimationFrame(loop);[cite: 2]
    }

    function startEngineClock() {
        const timeRing = document.getElementById("time-ring");
        const strokeMaxOffset = 660; // เส้นรอบวงของวงแหวนตามสูตรคำนวณ SVG

        setInterval(async () => {
            millisecondsLeft -= clockSpeed;

            if (millisecondsLeft <= 0) {
                millisecondsLeft = intervalPeriod; // รีเซ็ตตัวนับเวลากลับไปที่ 3 วินาที
                await predict(); // สั่งให้ AI ประมวลผลและทำนายวัตถุ[cite: 2]
            }

            // คำนวณความสั้น-ยาวของกราฟิกวงแหวนนับถอยหลังรอบกล้อง
            const updatedOffset = strokeMaxOffset - (strokeMaxOffset * (millisecondsLeft / intervalPeriod));
            timeRing.style.strokeDashoffset = updatedOffset;

            // พิมพ์ตัวเลขเวลาบนแถบ Header
            document.getElementById("timer-countdown").innerText = `NEXT CAPTURE: ${(millisecondsLeft / 1000).toFixed(1)}s`;
        }, clockSpeed);
    }

    async function predict() {[cite: 2]
        if (!model || !webcam) return;

        const prediction = await model.predict(webcam.canvas);[cite: 2]
        
        let maximumProbability = -1;
        let indexOfTheWinner = -1;

        // ค้นหาคลาสที่มีความเป็นไปได้สูงที่สุดในการสแกนรอบนั้นๆ
        for (let i = 0; i < maxPredictions; i++) {[cite: 2]
            if (prediction[i].probability > maximumProbability) {
                maximumProbability = prediction[i].probability;
                indexOfTheWinner = i;
            }
        }

        // นำข้อมูลลงไปแสดงผลในแถบ UI Card แต่ละช่อง
        for (let i = 0; i < maxPredictions; i++) {[cite: 2]
            const nameOfClass = prediction[i].className;
            const probabilityValue = prediction[i].probability;
            const roundedPercentage = Math.round(probabilityValue * 100);

            document.getElementById(`txt-name-${i}`).innerText = nameOfClass;
            document.getElementById(`txt-percent-${i}`).innerText = roundedPercentage + "%";
            document.getElementById(`bar-fill-${i}`).style.width = roundedPercentage + "%";

            // ดึง Element บล็อกมาเพื่อจัดการเอฟเฟกต์สี
            const cardBlock = document.getElementById(`class-box-${i}`);
            
            // ถ้าเป็นคลาสที่ชนะและมีความแม่นยำสูงเกิน 40% จะแสดงผลด้วยธีมไฮไลต์สีเขียวเรืองแสง
            if (i === indexOfTheWinner && roundedPercentage >= 40) {
                cardBlock.classList.add("active-match");
            } else {
                cardBlock.classList.remove("active-match");
            }
        }
    }
